# 8.1. MPA 방식의 링크

  • App.js
import './styles.css';
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import Diary from './pages/Diary';
import Edit from './pages/Edit';
import Home from './pages/Home';
import New from './pages/New';

export default function App() {
	return (
		<BrowserRouter>
			<div className='App'>
				<h1>Hello CodeSandbox</h1>
				<h2>Start editing to see some magic happen!</h2>
				<Routes>
					<Route path='/' element={<Home />} />
					<Route path='/new' element={<New />} />
					<Route path='/edit' element={<Edit />} />
					<Route path='/diary' element={<Diary />} />
				</Routes>
				//추가
				<a href={'/new'}>New</a>
			</div>
		</BrowserRouter>
	);
}

# 9.2. ② Query String (feat.useSearchParams)

  • 웹페이지에 데이터를 전달하는 가장 간단한 기법

  • /about?details=true 이런것들이 쿼리스트링

    네이버주소표시줄 스샷

useSearchParams 훅은 현재 위치에 대한 URL의 쿼리 문자열을 읽거나 수정할수 있다.
React의 useState 훅 처럼, 매개변수와 업데이트하는 함수라는 두 가지 값의 배열을 반환한다.

  7. useSearchParams 는 업데이트 함수를 제공하여 기존값을 변경할수 있다

    import { useSearchParams } from 'react-router-dom';
const style = { fontSize: '5rem', textAlign: 'center', backgroundColor: 'blue' };

const Edit = (props) => {
	const [params, setParams] = useSearchParams();
	const id = params.get('id');
	const mood = params.get('mood');
	const changeMood = () => setParams({ id: '50', mood: 'light' });
	return (
		<div>
			<h1 style={style}>{id} 번째 일기 수정</h1>
			<p>현재 {mood} 모드 상태입니다</p>
			<button onClick={changeMood}>{mood}</button>
		</div>
	);
};
export default Edit;

# 9.4. ③ Page Moving (feat.useNavigate)

useNavigate 훅은 프로그래밍 방식으로 탐색할 수 있는 함수를 반환한다.

  1. 홈이동 버튼 만들기

    //Edit.js
import { useNavigate, useSearchParams } from "react-router-dom";

const Edit = () => {
  const [searchParams, setSearchParams] = useSearchParams();
 const navigate=useNavigate();
 console.log(navigate);
 ...
}

    useNavigate 를 임포트 하고 변수에 할당후 콘솔창에 반환값을 확인해보자
    함수는 매개변수를 2개 받고 있고 첫번째는 to, 두번째는 option 으로 확인된다.
    공식 문서의 예문을 보면 to는 경로, option 은 객체형으로 다양한 설정값을 전달할수 있다.
